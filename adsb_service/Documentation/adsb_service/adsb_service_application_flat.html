	retrieve_auth_token: BOOLEAN
		local
			postfields: STRING_8
			res: INTEGER_32
			slist: POINTER
			useragent: STRING_8
			tk: STRING_8
			data: CURL_STRING
		do
			Result := False
			create tk.make_empty
			create data.make_empty
			create postfields.make_empty
			postfields := "{%"user%": %"" + auth_usr + "%",%"password%": %"" + auth_pwd + "%"}"
			create useragent.make_empty
			useragent := "adsb service"
			if is_api_available then
				curl_handle := init
				if not curl_handle.is_default_pointer then
					setopt_string (curl_handle, curlopt_url, auth_url)
					slist := slist_append (slist, "Accept:")
					slist := slist_append (slist, "Content-Type: application/json")
					setopt_slist (curl_handle, curlopt_httpheader, slist)
					setopt_string (curl_handle, curlopt_postfields, postfields)
					setopt_integer (curl_handle, curlopt_ssl_verifypeer, 0)
					setopt_integer (curl_handle, curlopt_ssl_verifyhost, 0)
					setopt_string (curl_handle, curlopt_useragent, useragent)
					setopt_integer (curl_handle, curlopt_fresh_connect, 1)
					setopt_integer (curl_handle, curlopt_forbid_reuse, 1)
					setopt_integer (curl_handle, curlopt_buffersize, 1024)
					set_write_function (curl_handle)
					setopt_curl_string (curl_handle, curlopt_writedata, data)
					res := perform (curl_handle)
					if res /= Curle_ok then
						logger.write_error ("cURL error " + res.out)
					else
						logger.write_information ("auth token request response: " + data)
						tk := data.split (':'.to_character_32).at (2);
						tk.prune_all ('"');
						tk.prune_all ('}');
						auth_token.set_token (tk);
						auth_token.set_expiry (create {DATE_TIME}.make_now end end + create {DATE_TIME_DURATION}.make_definite (1, 0, 0, 0) end);
						logger.write_information ("auth token: " + auth_token.token + " - " + auth_token.expiry.formatted_out ("yyyy/[0]mm/[0]dd [0]hh:[0]mi:[0]ss.ff3"))
						Result := True
					end
				end
			end
			cleanup (curl_handle)
		end

feature -- Query

	getinfo (a_curl_handle: POINTER; a_info: INTEGER_32; a_data: CELL [detachable ANY]): INTEGER_32
			-- `curl_getinfo
			-- (from CURL_EASY_EXTERNALS_I)
		require -- from CURL_EASY_EXTERNALS_I
			exists: not a_curl_handle.is_default_pointer
			valid: (create {CURL_INFO_CONSTANTS} end).is_valid (a_info)
		local
			l_api: POINTER
			mp: detachable MANAGED_POINTER
			l: INTEGER_32
			cs: C_STRING
			d: REAL_64
		do
			a_data.replace (Void)
			l_api := curl_easy_getinfo_ptr
			if a_info & {CURL_INFO_CONSTANTS}.curlinfo_long /= 0 then
				create mp.make ({PLATFORM}.integer_32_bytes)
			elseif a_info & {CURL_INFO_CONSTANTS}.curlinfo_string /= 0 then
				create mp.make ({PLATFORM}.pointer_bytes)
			elseif a_info & {CURL_INFO_CONSTANTS}.curlinfo_double /= 0 then
				create mp.make ({PLATFORM}.real_64_bytes)
			end
			if mp /= Void then
				Result := c_getinfo (l_api, a_curl_handle, a_info, mp.item)
				if Result = {CURL_CODES}.curle_ok then
					if a_info & {CURL_INFO_CONSTANTS}.curlinfo_long /= 0 then
						l := mp.read_integer_32 (0);
						a_data.put (l)
					elseif a_info & {CURL_INFO_CONSTANTS}.curlinfo_string /= 0 then
						create cs.make_shared_from_pointer (mp.read_pointer (0));
						a_data.put (cs.string)
					elseif a_info & {CURL_INFO_CONSTANTS}.curlinfo_double /= 0 then
						d := mp.read_real_64 (0);
						a_data.put (d)
					end
				end
			end
		end
