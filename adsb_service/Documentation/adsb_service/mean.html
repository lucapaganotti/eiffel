			-- write Current as a json object
		local
			l_filename: STRING_8
			l_mtext: STRING_8

		do
			if pos.count > 0 and then (source = ref_arpt or dest = ref_arpt) then

					create l_mtext.make_empty;
					l_mtext.append ("{");
					l_mtext.append ("%"A%":%"" + addr + "%",");

					if attached {STRING_8} callsign as l_cs then

					l_mtext.append ("%"D%":")

						l_mtext.append ("%"N.A.%"")
					end;
					l_mtext.append (",");
					l_mtext.append ("%"AT%":")
					if not type.is_empty then
						l_mtext.append (type)

						l_mtext.append ("%"Y%":" + pos.item.y.out + ",");
						l_mtext.append ("%"Z%":" + pos.item.z.out + ",");
						l_mtext.append ("%"T%":%"" + pos.item.t.out + "%",");
						l_mtext.append ("%"GS%":" + ground_speed.out + ",");
						l_mtext.append ("%"TA%":" + track_angle.out + ",");

												dest := "N.A."
											end
										else
											source := "N.A."
