	make_from_string (t: STRING_8)
		require
			t_attached: attached t
		do
			create hmap.make_map
			if t.has (Asterisk) then
				t.prune_all_leading (Asterisk)
			end
			if t.has (Semicolon) then
				t.prune_all_trailing (Semicolon)
			end
			text := t
			check
				text.count = Short_hex_msg_length or text.count = Long_hex_msg_length
			end
			create bin_text.make_empty
			to_bin
		ensure
			text_set: text = t
		end
