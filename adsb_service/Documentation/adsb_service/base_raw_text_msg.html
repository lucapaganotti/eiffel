class 
	BASE_RAW_TEXT_MSG

inherit
	RAW_MSG_CONSTANTS

	RAW_UTILS

create 
	make_from_string

feature {NONE} -- Initialization

	make_from_string (t: STRING_8)
		require
			t_attached: attached t
		do
			create hmap.make_map
			if t.has (Asterisk) then
				t.prune_all_leading (Asterisk)
			end
			if t.has (Semicolon) then
				t.prune_all_trailing (Semicolon)
			end
			text := t
			check
				text.count = Short_hex_msg_length or text.count = Long_hex_msg_length
			end
			create bin_text.make_empty
			to_bin
		ensure
			text_set: text = t
		end
	
feature -- Access

	text: STRING_8 assign set_text

	bin_text: STRING_8

	hmap: HEX_TO_BIN_MAP
	
feature -- Commands

	set_text (t: STRING_8)
		require
			t_attached: attached t
		do
			text := t
		ensure
			text_set: text = t
		end
	
feature {NONE} -- Operations

	to_bin
		local
			i: INTEGER_32
		do
			from
				i := 1
			until
				i = text.count
			loop
				if attached hmap.map.item (text.item (i)) as l_i then
					bin_text.append (l_i)
				end
			end
			check
				bin_text.count = Short_bin_msg_length or bin_text.count = Long_bin_msg_length
			end
		end
	
end -- class BASE_RAW_TEXT_MSG
